// Cloudflare Worker: anonymous feedback endpoint (/api/feedback) for nastyplot.org.
// Also /api/count: POST {"k":"teamcheck"} adds 1 to that day's total (D1 table counts); see feedback-lib.mjs.
// Accepts ONLY POST JSON {text <= 2000, page <= 40, kind <= 20} + Turnstile token in header `CF-Turnstile-Response` (or `X-Turnstile-Token`).
// Stores ONLY (at, page, kind, text) in D1 — NEVER IP, user-agent, headers, cookies, or viewer IDs.
// Only `export default` here: the helpers live in feedback-lib.mjs (Cloudflare accepts only entry points as named exports).
import {originAllowed, readCapped, scrubContact, verifyTurnstile, validateFeedback, validateCount, COUNT_MAX_BYTES, COUNT_SQL} from './feedback-lib.mjs';

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (url.pathname === '/api/feedback') {
      if (request.method !== 'POST') return new Response('Method Not Allowed', {status: 405});
      if (!originAllowed(request, env)) return Response.json({error: 'Forbidden origin'}, {status: 403});
      const raw = await readCapped(request);
      if (raw.tooBig) return Response.json({error: 'Body too large'}, {status: 413});
      let body;
      try { body = JSON.parse(raw.text); } catch (e) { return Response.json({error: 'Invalid JSON'}, {status: 400}); }
      const v = validateFeedback(body);
      if (v.error) return Response.json({error: v.error}, {status: 400});
      /* Workers rate limit (Nov 25: the free plan's one WAF rule is Cloudflare's leaked-credential check, so the limit lives here).
         The key is the sender's IP, held only by Cloudflare's limiter for its 60 s window; we never store or log it. No binding = no limit. */
      if (env.FEEDBACK_RL && typeof env.FEEDBACK_RL.limit === 'function') {
        const ip = request.headers.get('CF-Connecting-IP') || 'unknown';
        const {success} = await env.FEEDBACK_RL.limit({key: 'fb:' + ip});
        if (!success) return Response.json({error: 'Too many messages; try again in a minute'}, {status: 429});
      }
      const token = request.headers.get('CF-Turnstile-Response') || request.headers.get('X-Turnstile-Token') || '';
      const human = await verifyTurnstile(token, env.TURNSTILE_SECRET, fetch, env);
      if (!human) return Response.json({error: 'Verification failed'}, {status: 403});
      const at = new Date().toISOString().slice(0, 10);
      await env.DB.prepare('INSERT INTO feedback (at, page, kind, text) VALUES (?, ?, ?, ?)')
        .bind(at, v.ok.page, v.ok.kind, scrubContact(v.ok.text).text)
        .run();
      return Response.json({ok: true}, {status: 201});
    }
    // anonymous daily totals (Nov 25): POST {"k":"teamcheck"} only; stores (day, k) and n + 1, nothing else. No GET: no public stats.
    if (url.pathname === '/api/count') {
      if (request.method !== 'POST') return new Response('Method Not Allowed', {status: 405, headers: {Allow: 'POST'}});
      if (!originAllowed(request, env)) return Response.json({error: 'Forbidden origin'}, {status: 403});
      const raw = await readCapped(request, COUNT_MAX_BYTES);
      if (raw.tooBig) return Response.json({error: 'Body too large'}, {status: 413});
      let body;
      try { body = JSON.parse(raw.text); } catch (e) { return Response.json({error: 'Invalid JSON'}, {status: 400}); }
      const v = validateCount(body);
      if (v.error) return Response.json({error: v.error}, {status: 400});
      const day = new Date().toISOString().slice(0, 10);
      await env.DB.prepare(COUNT_SQL).bind(day, v.ok.k).run();
      return new Response(null, {status: 204});
    }
    if (url.pathname.startsWith('/data/')) {
      const res = await env.ASSETS.fetch(request);
      return res.status === 404 ? new Response('Not Found', {status: 404}) : res;
    }
    return env.ASSETS.fetch(request);
  },
};

=================== FILE worker/feedback-lib.mjs
// Feedback Worker helpers (Nov 25): split out of feedback.mjs because Cloudflare treats every named export of the
// Worker's main file as an entry point and refuses plain values (wrangler dev: "Incorrect type for map entry 'MAX_BODY_BYTES'").
// Cloudflare Worker: anonymous feedback endpoint (/api/feedback) for nastyplot.org.
// Accepts ONLY POST JSON {text <= 2000, page <= 40, kind <= 20} + Turnstile token in header `CF-Turnstile-Response` (or `X-Turnstile-Token`).
// Stores ONLY (at, page, kind, text) in D1 — NEVER IP, user-agent, headers, cookies, or viewer IDs.
const ALLOWED_KINDS = new Set(['bug', 'confusing', 'idea', 'like', 'mistake']);

// ---------- server-side guards (Nov 25): size, origin, and no contact details kept ----------
// Size: the body is read as text with a hard cap BEFORE JSON parsing, so a huge or never-ending body is refused early
// (Content-Length can be missing or wrong, so it is checked and then the real byte count is checked too). 16 KB is far above
// the largest valid body (2,000 characters of text, up to 4 bytes each in UTF-8, plus the JSON around it).
export const MAX_BODY_BYTES = 16384;
// Origin: browsers send Origin on every POST, so a request from another site carries that site's Origin and is refused.
// A request with no Origin at all (not a browser) still needs a valid Turnstile token. Set REQUIRE_ORIGIN = "1" to refuse
// those too. ALLOWED_ORIGINS (comma-separated) overrides the list below.
export const DEFAULT_ORIGINS = ['https://nastyplot.org', 'https://www.nastyplot.org'];
export function originAllowed(request, env = {}) {
  const list = typeof env.ALLOWED_ORIGINS === 'string' && env.ALLOWED_ORIGINS.trim()
    ? env.ALLOWED_ORIGINS.split(',').map(s => s.trim().replace(/\/+$/, '')).filter(Boolean) : DEFAULT_ORIGINS;
  const origin = request.headers.get('Origin'), site = request.headers.get('Sec-Fetch-Site');
  if (site && site !== 'same-origin' && site !== 'none') return false; // a browser says another site sent it
  if (origin == null || origin === '') return env.REQUIRE_ORIGIN !== '1';
  return list.includes(origin.replace(/\/+$/, ''));
}
export async function readCapped(request, max = MAX_BODY_BYTES) {
  const len = request.headers.get('Content-Length');
  if (len != null && len !== '' && (!/^\d+$/.test(len) || +len > max)) return {tooBig: true};
  if (!request.body) return {text: ''};
  const reader = request.body.getReader(), parts = []; let n = 0;
  for (;;) { const {done, value} = await reader.read(); if (done) break;
    n += value.byteLength; if (n > max) { try { await reader.cancel(); } catch (e) {} return {tooBig: true}; }
    parts.push(value); }
  const buf = new Uint8Array(n); let o = 0; for (const p of parts) { buf.set(p, o); o += p.byteLength; }
  return {text: new TextDecoder().decode(buf)};
}
// Contact details: kids use the site, so feedback never stores an email address or a phone number, even one typed into the
// text by mistake. They are replaced before storing; everything else in the message is kept as typed.
// Emails: plain (name@site.com) and spelled out (name at gmail dot com, name(at)site(dot)org, name [at] site [dot] net).
const TLD = '(?:com|net|org|edu|gov|io|co|uk|us|ca|au|de|fr|es|it|nl|jp|br|me|info|biz|app|dev|gg)';
const AT = '(?:@|\\s*[\\(\\[\\{]\\s*at\\s*[\\)\\]\\}]\\s*|\\s+at\\s+)', DOT = '(?:\\.|\\s*[\\(\\[\\{]\\s*dot\\s*[\\)\\]\\}]\\s*|\\s+dot\\s+)';
const EMAIL_RE = new RegExp(`[a-z0-9][a-z0-9._%+-]*${AT}[a-z0-9-]+(?:${DOT}[a-z0-9-]+)*${DOT}${TLD}\\b`, 'gi');
const EMAIL_PLAIN_RE = /[a-z0-9._%+-]+@[a-z0-9-]+(?:\.[a-z0-9-]+)+/gi;
// Phones: a run of digits with phone separators (space, dash, dot, brackets, a leading +) holding 10 to 15 digits, or a
// local 7-digit number written 555-1234 / 555.1234. Commas and slashes are not phone separators, so "13,631", "252/4/0" and
// stat spreads stay. A run of plain numbers with spaces can look like a phone and is removed too: losing a number is safer
// than keeping a phone number.
const PHONE_RE = /(?:\+\s?)?(?:\(\s?\d+\s?\)[\s.-]?)?\d(?:[\s.()-]{0,3}\d){9,14}(?!\d)/g;
const PHONE7_RE = /(?<![\d.,/])\b\d{3}[-.]\d{4}\b(?![\d.,/%])/g;
export function scrubContact(text) {
  let t = String(text || ''), emails = 0, phones = 0;
  t = t.replace(EMAIL_RE, () => { emails++; return '[email removed]'; }).replace(EMAIL_PLAIN_RE, () => { emails++; return '[email removed]'; });
  t = t.replace(PHONE_RE, m => { const d = m.replace(/\D/g, '').length; if (d < 10 || d > 15) return m; phones++; return '[phone removed]'; });
  t = t.replace(PHONE7_RE, () => { phones++; return '[phone removed]'; });
  return {text: t, emails, phones};
}

// Turnstile (launch review S1/S2, Nov 25): a token only counts if Cloudflare says it is good AND it was made on our own site for
// the feedback form (hostname + action), so a token solved on another site that uses our public site key is refused.
export const TS_ACTION = 'feedback';
export const DEFAULT_TS_HOSTS = ['nastyplot.org', 'www.nastyplot.org'];
export function turnstileOk(j, env = {}) {
  const hosts = typeof env.TURNSTILE_HOSTNAMES === 'string' && env.TURNSTILE_HOSTNAMES.trim()
    ? env.TURNSTILE_HOSTNAMES.split(',').map(s => s.trim()).filter(Boolean) : DEFAULT_TS_HOSTS;
  return !!(j && j.success === true && hosts.includes(j.hostname) && j.action === TS_ACTION);
}
export async function verifyTurnstile(token, secret, fetchImpl = fetch, env = {}) {
  if (!token || typeof token !== 'string' || token.length > 2048 || !secret) return false;
  const r = await fetchImpl('https://challenges.cloudflare.com/turnstile/v0/siteverify', {
    method: 'POST',
    headers: {'Content-Type': 'application/x-www-form-urlencoded'},
    body: new URLSearchParams({secret, response: token}).toString(),
  });
  if (!r.ok) return false;
  const j = await r.json();
  return turnstileOk(j, env);
}

export function validateFeedback(body) {
  if (!body || typeof body !== 'object' || Array.isArray(body)) return {error: 'Invalid JSON body'};
  const keys = Object.keys(body);
  if (keys.some(k => k !== 'text' && k !== 'page' && k !== 'kind')) return {error: 'Unexpected fields'};
  const text = typeof body.text === 'string' ? body.text.trim() : '';
  const page = typeof body.page === 'string' ? body.page.trim() : '';
  const kind = typeof body.kind === 'string' ? body.kind.trim() : '';
  if (text.length < 3 || text.length > 2000) return {error: 'text must be 3..2000 chars'};
  if (!page || page.length > 40) return {error: 'page must be 1..40 chars'};
  if (!kind || kind.length > 20 || !ALLOWED_KINDS.has(kind)) return {error: 'invalid kind'};
  return {ok: {text, page, kind}};
}


// ---------- anonymous counts (/api/count, Nov 25; queue item 4: marketing's baseline from the quiet launch) ----------
// The page sends {"k":"teamcheck"} when a full team gets its Strength Score (once per distinct six per page load, decided in the
// browser). Only a daily total is kept: one row per (day, event name) with n = n + 1. No IP, user agent, headers, cookies, referrer,
// time of day or any id is read or stored. Event names come from this allow-list; anything else is refused.
export const COUNT_KEYS = new Set(['teamcheck']);
// the largest valid body is 17 bytes ({"k":"teamcheck"}); 256 leaves room for spaces and a longer future name, nothing more
export const COUNT_MAX_BYTES = 256;
export function validateCount(body) {
  if (!body || typeof body !== 'object' || Array.isArray(body)) return {error: 'Invalid JSON body'};
  const keys = Object.keys(body);
  if (keys.length !== 1 || keys[0] !== 'k') return {error: 'Unexpected fields'};
  if (typeof body.k !== 'string' || !COUNT_KEYS.has(body.k)) return {error: 'invalid k'};
  return {ok: {k: body.k}};
}
export const COUNT_SQL = 'INSERT INTO counts (day, k, n) VALUES (?, ?, 1) ON CONFLICT(day, k) DO UPDATE SET n = n + 1';

// Keeps every daily data document under the page's storage limit (262,144 bytes per document) without dropping anything.
// Nov 10: meta/results grew past it (263,734 bytes, about +3 KB a day) and the morning refresh was refused, so the site kept
// yesterday's winning cores and proven teams. Model (229 KB) and events (232 KB) were close behind.
// For each document the refresh writes, if its JSON is over LIMIT (200,000 bytes, leaving room to grow) it is split in two:
// whole top-level keys, or the tail of the biggest array, move to part 2 until part 1 fits. Both parts carry the same
// _part stamp; the page (03-persistence.js metaJoin) joins them back in order. Writes out/split/<doc>.json + <doc>_2.json and
// out/split/index.json (which documents are split, their sizes). The source files are not changed (tests read them as they are).
// Exits 1 (and the Data health step reports it) if a document would need more than two parts.
import fs from 'node:fs';
const LIMIT = 200000, HARD = 262144, B = x => Buffer.byteLength(JSON.stringify(x));
const rd = f => { try { return JSON.parse(fs.readFileSync('out/' + f, 'utf8')); } catch (e) { return null; } };
// the page document <- the file the morning refresh reads (same list as the refresh's step 3; results = limitless.json "results")
const DOCS = {results: () => (rd('limitless.json') || {}).results, doubles_tour: () => rd('meta_tour.json'), teams: () => rd('teams.json'), history: () => rd('history.json'),
  model: () => rd('model.json'), doubles: () => rd('meta_ladder.json'), doubles_ingame: () => rd('meta_ingame.json'), events: () => rd('events.json'), track: () => rd('track.json'),
  practice: () => rd('practice.json'), weekly: () => rd('weekly.json'), calendar: () => rd('calendar.json'),
  post: () => rd('post.json') /* Nov 18: Mystery Gifts and news (post-news.mjs); about 22 KB, far under the limit */};
export const DOC_SOURCES = DOCS; /* Nov 25: the public site's deploy (public-site.mjs) writes the same documents as files */
export function split(doc, limit = LIMIT) {
  const a = {...doc}, b = {}; let guard = 0;
  while (B(a) > limit && guard++ < 200) {
    const keys = Object.keys(a).filter(k => k !== '_part').sort((x, y) => B(a[y]) - B(a[x])); const k = keys[0]; if (!k) break;
    const over = B(a) - limit, v = a[k];
    if (Array.isArray(v) && v.length > 1 && B(v) > over) { // move the tail of this array, just enough
      let n = Math.max(1, Math.ceil(v.length * over / B(v))); while (n < v.length && B(v.slice(0, v.length - n)) > B(v) - over - 2) n++;
      b[k] = v.slice(v.length - n).concat(b[k] || []); a[k] = v.slice(0, v.length - n);
    } else if (v && typeof v === 'object' && !Array.isArray(v) && Object.keys(v).length > 1 && B(v) > over) { // move some keys of this object
      const inner = Object.keys(v).sort((x, y) => B(v[y]) - B(v[x])), keep = {...v}, mv = {}; for (const ik of inner) { if (B(keep) <= B(v) - over - 2) break; mv[ik] = keep[ik]; delete keep[ik]; }
      a[k] = keep; b[k] = {...(b[k] || {}), ...mv};
    } else { b[k] = v; delete a[k]; } }
  return {a, b};
}
// the page's join (kept identical to joinParts in app/src/app/03-persistence.js), used here to prove the split loses nothing
export function join(a, b) { const o = {...a}; for (const k in b) { if (k === '_part') continue; const x = o[k], y = b[k]; o[k] = Array.isArray(x) && Array.isArray(y) ? x.concat(y) : x && y && typeof x === 'object' && typeof y === 'object' && !Array.isArray(x) ? {...x, ...y} : y; } delete o._part; return o; }
// same content regardless of key order (moving a key to part 2 changes where it sits in the object, not what it holds)
export const canon = x => JSON.stringify(x, (k, v) => v && typeof v === 'object' && !Array.isArray(v) ? Object.fromEntries(Object.keys(v).sort().map(q => [q, v[q]])) : v);
if (process.argv[1] && process.argv[1].endsWith('split-docs.mjs')) {
  fs.rmSync('out/split', {recursive: true, force: true}); fs.mkdirSync('out/split', {recursive: true});
  const index = {limit: LIMIT, hard: HARD, made: new Date().toISOString(), split: {}, sizes: {}}; let bad = 0;
  for (const [name, get] of Object.entries(DOCS)) { const doc = get(); if (!doc || typeof doc !== 'object') continue; const n = B(doc); index.sizes[name] = n;
    if (n <= LIMIT) continue;
    const {a, b} = split(doc), id = `${name}-${index.made}`; a._part = {id, n: 1}; b._part = {id, n: 2};
    const same = canon(join(a, b)) === canon(doc), sa = B(a), sb = B(b);
    if (!same || sa > HARD || sb > HARD) { bad++; console.log(`::error::split-docs: ${name} (${n} bytes) can't be split into two parts that fit (${sa} + ${sb}, joins back the same: ${same})`); continue; }

}
