    const b = assignItems([mk(pair[1]), mk(pair[0])]).map(m => m.p.n + ':' + m.item).sort().join('|');
    if (a !== b) throw new Error(`J11-1: assignItems order-dependent for ${pair.join(' vs ')}: ${a} !== ${b}`);
  });

  // Case J11-2 (Invariant 5): Item Clause clash sets G.illegal = true; dead item alone does not (with Diff B)
  // S7 fails if: Item Clause stops setting G.illegal = true, or (with Diff B) a legal dead-item team sets G.illegal = true.
  await page.evaluate(() => {
    const nonMegas = ROSTER.filter(p => !p.m).slice(0, 5);
    const mega = ROSTER.find(p => p.m);
    const mk = (p, it) => ({ i: 0, p, n: p.n, d: p.d, types: p.t, ability: p.a[0], mega: p.m, base: p.b, moves: popularMoves(p).slice(0, 4), heldItem: p.m ? null : it, item: p.m ? stoneFor(p) : it });
    const clashTeam = [mk(mega, null), mk(nonMegas[0], 'Sitrus Berry'), mk(nonMegas[1], 'Sitrus Berry'), mk(nonMegas[2], 'Leftovers'), mk(nonMegas[3], 'Focus Sash'), mk(nonMegas[4], 'Rocky Helmet')];
    const RD = THR.teamReport(clashTeam);
    const G = CH.gradeOf(RD, proChecks(clashTeam, true), clashTeam);
    if (!G || !G.illegal) throw new Error('J11-2: Item Clause clash did not set G.illegal = true');
    const deadTeam = [mk(mega, null), mk(nonMegas[0], 'Psychic Seed'), mk(nonMegas[1], 'Sitrus Berry'), mk(nonMegas[2], 'Leftovers'), mk(nonMegas[3], 'Focus Sash'), mk(nonMegas[4], 'Rocky Helmet')];
    const hasPsy = deadTeam.some(m => TERRAIN_OF[m.ability] === 'psy');
    if (!hasPsy) {
      const GD = CH.gradeOf(THR.teamReport(deadTeam), proChecks(deadTeam, true), deadTeam);
      if (GD && GD.illegal && itemClash(deadTeam).length === 0) {
        throw new Error('J11-2b: legal team with dead Psychic Seed was flagged G.illegal = true');
      }
    }
  });

  // Case J11-3 (Invariant 1 + Speed): THR.teamReport is order-independent and duelCache matches THR.noCache(true)
  // S7 fails if: reordering a 6-Pokémon team changes THR.teamReport().score, or duelCache diverges from uncached.
  await page.evaluate(() => {
    const six = ROSTER.slice(0, 6).map(p => ({ i: 0, p, n: p.n, d: p.d, types: p.t, ability: p.a[0], mega: p.m, base: p.b, moves: popularMoves(p).slice(0, 4), heldItem: null, item: '' }));
    const msA = assignItems(six);
    const msB = assignItems(six.slice().reverse());
    const rA = THR.teamReport(msA);
    const rB = THR.teamReport(msB);
    if (rA.score !== rB.score) throw new Error(`J11-3a: THR.teamReport order-dependent: ${rA.score} !== ${rB.score}`);
    THR.noCache(true);
    const rUncached = THR.teamReport(msA);
    THR.noCache(false);
    if (rA.score !== rUncached.score) throw new Error(`J11-3b: duelCache mismatch: cached ${rA.score} !== uncached ${rUncached.score}`);
  });

  // Case J11-4 (Invariant 3 / NaN guard): THR.report stays finite if a threat weight is non-finite (with Diff A)
  // S7 fails if: a single NaN/missing threat weight poisons reportSidesG().score to NaN.
  await page.evaluate(() => {
    const a = ROSTER[0].n, b = ROSTER[1].n, t1 = ROSTER[2].n, t2 = ROSTER[3].n;
    const rep = THR.report([{ n: a }, { n: b }], [{ n: t1, w: 10 }, { n: t2, w: NaN }]);
    if (!Number.isFinite(rep.score)) throw new Error(`J11-4: THR.report returned non-finite score ${rep.score} on NaN threat weight`);
  });

  // Case J11-5 (Invariant 9): Share image ARB.score matches CH.gradeNum / Math.round(G.s) for full 6-Pokémon teams
  // S7 fails if: ARB.score(G).num diverges from Math.round(G.s) on a full 6-Pokémon team.
  await page.evaluate(() => {
    const six = assignItems(ROSTER.slice(0, 6).map(p => ({ i: 0, p, n: p.n, d: p.d, types: p.t, ability: p.a[0], mega: p.m, base: p.b, moves: popularMoves(p).slice(0, 4), heldItem: null, item: '' })));
    const f = ARB.facts(six, 'Parity Test');
    const sc = ARB.score(f.G);
    if (!f.G || !sc) throw new Error('J11-5: ARB.facts/ARB.score returned null for a 6-Pokémon team');
    if (f.G.v === 2 && sc.num !== String(Math.round(f.G.s))) {
      throw new Error(`J11-5: share image score ${sc.num} !== page score ${Math.round(f.G.s)}`);
    }
  });
}

--- ORDER QUESTION 1 (Job 11 #6): The `89.6 -> "90 · Tournament ready"` band-edge case ---

Currently (`58-strength.js:143, 214, 223`), `s = Math.round(st.ss * 10) / 10` (e.g., `89.6`), while
`CH.gradeFull(s)` and `CH.wordOf(s)` read `89.6` ('B' / 'Tournament ready') and the UI/share card print
`Math.round(G.s)` (`90`). That puts `"90 · Tournament ready"` right above `CH.LETTERS` (`58-strength.js:221`:
`"90+ Top tier · 80s Tournament ready..."`).

Options:
(a) Letter and word from the rounded number (`Math.round(s)`) — what the player sees.
    - Matches: `58-strength.js:118` ("The letter comes from the score as shown, rounded"), `58-strength.js:221`
      (`CH.LETTERS`), and `CLAUDE.md` #19 ("Every number shown is honest").
    - Trade-off: effective band edge shifts down 0.5 pts (`89.5 -> 90 · Top tier`); `CH.sticky` should
      also evaluate against `Math.round(s)` so sticky doesn't re-create a mismatch.
(b) Show one decimal at the band edge (e.g., `89.6 · Tournament ready`).
    - Matches: `58-strength.js:23-26` (exact `>= 90` thresholds) and `58-strength.js:143` (1-decimal `s`).
    - Trade-off: breaks integer-only school-grade formatting (`06-helpers.js:2, 212`: "10 or more -> whole
      number") and mixes `89.6` with `89` / `90` across teams.
(c) Floor (`Math.floor(G.s)`) instead of `Math.round(G.s)` (`89.6 -> "89 · Tournament ready"`).
    - Matches: `58-strength.js:23-26` (strict `>= 90` for `90+`) AND integer-only display (`06-helpers.js:212`)
      with zero contradiction against `CH.LETTERS`.
    - Trade-off: displays ~0.5 pts lower on average (`89.9 -> 89`); `CH.whyData` (`base + mu + own = t`)
      and `CH.gradeMove` would need to use `Math.floor` consistently so parts still add up.

Recommendation: Option (a) matches the existing comment at `58-strength.js:118` and requires the smallest
change; Option (c) is cleanest if `>= 90.0` must stay strict.
