// Thin-data counts for the active regulation (Phase 0.8, Nov 24): writes out/regstate.json
// ({reg, day, events, players, games, since, updated}) so pages can show "Day N of Reg M-D: X events so far".
import fs from 'node:fs';
import {pathToFileURL} from 'node:url';
import {activeReg, readReg} from './formats.mjs';

const rd = f => { try { return JSON.parse(fs.readFileSync(f, 'utf8')); } catch (e) { return null; } };

export function buildRegState({regObj, rawReg, matches, today} = {}) {
  const RG = regObj || activeReg();
  const R = rawReg !== undefined ? rawReg : readReg();
  const d = today || new Date().toISOString().slice(0, 10);
  const since = (RG && RG.since) || d;
  const t0 = Date.parse(since + 'T12:00:00Z'), t1 = Date.parse(d + 'T12:00:00Z');
  const day = isFinite(t0) && isFinite(t1) ? Math.max(1, Math.floor((t1 - t0) / 864e5) + 1) : 1;
  const mOk = matches && (!matches.reg || matches.reg === RG.id);
  const seen = R && R.seen && R.seen[RG.id];
  const events = mOk && Array.isArray(matches.events) ? matches.events.length : (seen ? +seen.events || 0 : 0);
  const players = mOk && typeof matches.players === 'number' ? matches.players : (seen ? +seen.players || 0 : 0);
  const games = mOk && Array.isArray(matches.m) ? matches.m.length : 0;
  return {reg: RG.id, day, events, players, games, since, updated: new Date().toISOString()};
}

if (import.meta.url === pathToFileURL(process.argv[1] || '').href) {
  const out = buildRegState({matches: rd('out/matches.json')});
  fs.writeFileSync('out/regstate.json', JSON.stringify(out));
  console.log(`regstate: ${out.reg} day ${out.day} (since ${out.since}), ${out.events} events, ${out.players} players, ${out.games} games`);
}

6. ORDER QUESTIONS, "I'M UNSURE", AND WHAT WAS NOT LOOKED AT
====================================================================

--- Order questions for HH (S1) ---
1. `regstate.mjs` day numbering: `day: 1` on `since` (e.g., 2026-12-02 UTC = Day 1, 2026-12-04 UTC = Day 3) vs `day: 0` on `since`.
   - Option A (used here): 1-indexed (`Day 1 of Reg M-D` on launch day, matching `inLaunch: day 1..14`).
   - Option B: 0-indexed (`day: 0` on launch day).
2. `leads.mjs` when `MD.ships === false` (early M-D):
   - Option A (used here): still runs if `replays.json` is M-D (`rReg === mdReg === curReg`), writing M-D `leads` (or `{}` if <150 games) into `model.json`.
   - Option B: skip `leads.mjs` whenever `!MD.ships` so `leads` only ships alongside a shipped model.
3. `track.json` M-C archive location:
   - Option A (used here): `out/track.json` holds only active reg (staying under the 240 KB Firestore cap) and archives M-C to `out/track_M-C.json`.
   - Option B: store `{byReg: {"M-C": ..., "M-D": ...}}` inside `out/track.json` (risks hitting the 240 KB / 262,144-byte Firestore cap sooner).

- I'm unsure whether `app/`'s `META.model` reader (`13-rating.js` / team grader) checks `META.model && META.model.ships` or only `META.model && META.model.mons && Object.keys(META.model.mons).length`, because `app/` files were not included in Job 8. Including `ships: false, mons: {}` in the not-ready `model.json` is safe for `six.mjs` and `ctr.mjs`, and should be checked against `app/` before merge.
- I'm unsure whether `pilot.mjs` (`pilotTest(M)`) or `team-basics.mjs` (`deckBasics`) handle small non-empty `matches.json` (e.g. 5 events, 40 games) without throwing if `model.mjs` reaches line 243, because neither file was included in Job 8.
