    const opts = {
      level: 50, nature: s.nat, ability: s.ability || '(none)',
      item: role === 'a' && p.m && !s.item ? 'Leftovers' : (s.item || undefined),

    };
    return new Pokemon(G0, name, opts);
  };
  const A = mk(c.a, 'a'), D = mk(c.d, 'd');
  const move = new Move(G0, c.mv, {
    ability: A.ability,

  });
  const W = {sun: 'Sun', rain: 'Rain', sand: 'Sand', snow: 'Snow'}[c.weather];

  const field = new Field({gameType: 'Doubles', weather: W, terrain: TR});
  const r = calculate(G0, A, D, move, field);
  let d = r.damage;
  if (typeof d === 'number') d = Array(16).fill(d);
  else if (Array.isArray(d[0])) d = d[0].map((_, i) => d.reduce((s, h) => s + h[i], 0));

}

// Cloudflare Worker: anonymous feedback endpoint (/api/feedback) for nastyplot.org.
// Accepts ONLY POST JSON {text <= 2000, page <= 40, kind <= 20} + Turnstile token in header `CF-Turnstile-Response` (or `X-Turnstile-Token`).
// Stores ONLY (at, page, kind, text) in D1 — NEVER IP, user-agent, headers, cookies, or viewer IDs.
const ALLOWED_KINDS = new Set(['bug', 'confusing', 'idea', 'like', 'mistake']);

export async function verifyTurnstile(token, secret, fetchImpl = fetch) {
  if (!token || typeof token !== 'string' || !secret) return false;
  const r = await fetchImpl('https://challenges.cloudflare.com/turnstile/v0/siteverify', {
    method: 'POST',
    headers: {'Content-Type': 'application/x-www-form-urlencoded'},
    body: new URLSearchParams({secret, response: token}).toString(),
  });
  if (!r.ok) return false;
  const j = await r.json();
  return !!(j && j.success);
}

export function validateFeedback(body) {
  if (!body || typeof body !== 'object' || Array.isArray(body)) return {error: 'Invalid JSON body'};
  const keys = Object.keys(body);
  if (keys.some(k => k !== 'text' && k !== 'page' && k !== 'kind')) return {error: 'Unexpected fields'};
  const text = typeof body.text === 'string' ? body.text.trim() : '';
  const page = typeof body.page === 'string' ? body.page.trim() : '';
  const kind = typeof body.kind === 'string' ? body.kind.trim() : '';
  if (text.length < 3 || text.length > 2000) return {error: 'text must be 3..2000 chars'};
  if (!page || page.length > 40) return {error: 'page must be 1..40 chars'};
  if (!kind || kind.length > 20 || !ALLOWED_KINDS.has(kind)) return {error: 'invalid kind'};
  return {ok: {text, page, kind}};
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (url.pathname === '/api/feedback') {
      if (request.method !== 'POST') return new Response('Method Not Allowed', {status: 405});
      let body;
      try { body = await request.json(); } catch (e) { return Response.json({error: 'Invalid JSON'}, {status: 400}); }
      const v = validateFeedback(body);
      if (v.error) return Response.json({error: v.error}, {status: 400});
      const token = request.headers.get('CF-Turnstile-Response') || request.headers.get('X-Turnstile-Token') || '';
      const human = await verifyTurnstile(token, env.TURNSTILE_SECRET);
      if (!human) return Response.json({error: 'Verification failed'}, {status: 403});
      const at = new Date().toISOString().slice(0, 10);
      await env.DB.prepare('INSERT INTO feedback (at, page, kind, text) VALUES (?, ?, ?, ?)')
        .bind(at, v.ok.page, v.ok.kind, v.ok.text)
        .run();
      return Response.json({ok: true}, {status: 201});
    }
    if (url.pathname.startsWith('/data/')) {
      const res = await env.ASSETS.fetch(request);
      return res.status === 404 ? new Response('Not Found', {status: 404}) : res;
    }
    return env.ASSETS.fetch(request);
  },
};

export function auditRouteSnapshot({pageErrors = [], text = '', h1Text = ''}) {
  const errs = [];
  if (pageErrors.length) errs.push('pageerror: ' + pageErrors.join(' | '));
  const m = String(text || '').match(BAD_TEXT_RE);
  if (m) errs.push('bad visible text: ' + m[0]);
  if (!String(h1Text || '').trim()) errs.push('empty or missing &lt;h1&gt;');
  return errs;
}

## 3. Regulation rules & the date switch

- **Single source of truth:** `regulation.json` holds `"site"` (`"M-C"`) and each regulation's entry (`name`, `sd`, `since`, `end`, `sources`, `items`, `rules`).
- **Off-format filtering (`formats.mjs`):**
  - `OFF_FORMAT` regex skips themed/restricted cups by name (Spooky, Monotype, Monoregion, UU, Gen 1–8, Draft, Little Cup, etc.). Raw files stay in `data/`.
  - `oddEvents()` automatically excludes events with 20+ teams missing a 20%+ staple (expected on 8+ teams) or 8+ teams whose top-10 staple count is under half normal.
- **Official end dates (`REG_END` / `regAt`):** Limitless sometimes tags events after a regulation ends with the old tag. Any event dated at or after `REG_END[id]` (`2026-12-02T01:59:00Z` for M-C) is reclassified to `nextRegIds(id)[0]` (`M-D`). `update.mjs` also checks the previous format's recent pages for such mis-tagged events.
- **Date-based switch (`detectReg` in `formats.mjs`):**
  - When `nextProf = regProfile(nextRegIds(id)[0])` has a `since` date (e.g., `2026-12-02` for M-D), `reg.mjs` switches on launch day (`today UTC >= nextProf.since`), even if Limitless/Showdown are unreachable that morning (`dateDue`).
  - `REG_SWITCH` (`{events: 20, players: 1200}`) stays as a backstop only when the next regulation has no `since` date in `regulation.json`.
  - Setting env `REG_AUTO_SWITCH=0` holds the switch (shows a banner only) for manual review.
- **Outside-source guard (`sources.mjs` + `formats.mjs`):** Null source IDs in `regulation.json` log `"not available yet"` instead of reading old-regulation pages. `sourcePageCheck()` verifies Showdowntier names `Regulation <id>` and Pikalytics' sample events are at least half `<id>`.

---

## 4. The regulation guard (`reg-guard.mjs`)

- Runs **twice** in `update.yml`: once before `perf.mjs` (so `perf.json` cannot copy stale-regulation inputs) and once before `split-docs.mjs` (so `out/split/` never ships mixed regulations).
- **Never touches `data/`** (all raw tournament standings/pairings stay intact).
- **Exempt files** (never emptied): `reg.json`, `calendar.json`, `skips.json`, `official_perf.json`, `ratings_log.json` (per-reg `byReg`), `scoreboard.json` (multi-reg history).
- **Archive first:** Before emptying a mismatched file, `reg-guard.mjs` copies it once to `out/archive/<old-reg>/<file>` so no old regulation data is ever lost.
- **Empty shapes:** Replaces each mismatched (or unstamped post-switch) file with `emptyFor(file, reg, sd)` so the page and `health.mjs` (`thin` mode, days 1–14) handle day one cleanly.
- **Never fails the daily job:** Errors log a `::warning::` and exit 0.

- **`md-watch.mjs`** (run daily by `.github/workflows/md-watch.yml`, read-only):
  - Checks 3 places for the next regulation (`M-D`): Showdown `config/formats.ts`, Limitless VGC page 1 (with one 10 s retry), and two official news JSON feeds (`champions-news.pokemon-home.com` and `news.pokemon-home.com`). Opens one issue per sighting (`"Reg M-D spotted: <where>"`).
  - **Limitless pause alarm:** Opens `"Limitless data paused since <date>"` if Limitless fails both tries or `data/index.json`'s newest tournament is 2+ days old; closes automatically when data resumes.

// Google structured data (JSON-LD) for nastyplot.org public builds (ZERO visible words).
// Values come strictly from docs/rebuild/seo-titles-draft.md and the route list.
// Note: SearchAction is intentionally omitted because the public site has no /?q= search URL (search is an in-page modal).
export const SITE_ORIGIN = 'https://nastyplot.org';

export function websiteJsonLd(homeRow = null) {
  const desc = (homeRow && homeRow.desc) ||
    'Build a winning Pokémon Champions team: Strength Score, damage calc, speed tiers, usage stats and winning tournament teams, updated every day.';
  return {
    '@context': 'https://schema.org',
    '@type': 'WebSite',
    name: 'NastyPlot',
    url: SITE_ORIGIN + '/',
    description: desc,
  };
}

export function pokemonBreadcrumbJsonLd(name, displayName = '') {
  const clean = String(name || '').trim();
  if (!clean) return null;
  const label = String(displayName || clean).trim();
  return {
    '@context': 'https://schema.org',
    '@type': 'BreadcrumbList',
    itemListElement: [
      {
        '@type': 'ListItem',
        position: 1,
        name: 'NastyPlot',
        item: SITE_ORIGIN + '/',
      },
      {
        '@type': 'ListItem',
        position: 2,
        name: 'Ratings',
        item: SITE_ORIGIN + '/meta',
      },
      {
        '@type': 'ListItem',
        position: 3,
        name: label,
        item: SITE_ORIGIN + '/pokemon/' + encodeURIComponent(clean),
      },
    ],
  };
}

export function pokemonPageGraphJsonLd(name, displayName = '', homeRow = null) {
  const ws = websiteJsonLd(homeRow);
  const bc = pokemonBreadcrumbJsonLd(name, displayName);
  if (!bc) return ws;
  const {'@context': _, ...wsBody} = ws;
  const {'@context': __, ...bcBody} = bc;
  return {
    '@context': 'https://schema.org',
    '@graph': [wsBody, bcBody],
  };
}

export function jsonLdScriptTag(obj) {
  const json = JSON.stringify(obj).replace(/</g, '\\u003c');
  return `<script type="application/ld+json" id="np-jsonld">${json}</script>`;
}

// Schema.org JSON-LD validation for the public build (Node 22, zero dependencies).
// Usage: node experiments/jsonld-test.mjs
import fs from 'node:fs';
import {
  SITE_ORIGIN, websiteJsonLd, pokemonBreadcrumbJsonLd, pokemonPageGraphJsonLd, jsonLdScriptTag,
} from '../app/public-jsonld.mjs';

let failed = 0;
const check = (id, ok, detail = '') => {
  console.log(`${ok ? 'PASS' : 'FAIL'} ${id}${detail ? ': ' + detail : ''}`);
  if (!ok) failed++;
};

const isHttpsUrl = s => typeof s === 'string' && /^https:\/\/nastyplot\.org(\/|$)/.test(s);

// LD-1: Read home row from docs/rebuild/seo-titles-draft.md (values strictly from existing data)
let homeDesc = '';
{
  const lines = fs.readFileSync('docs/rebuild/seo-titles-draft.md', 'utf8').split('\n');
  for (const line of lines) {
    if (!line.startsWith('| home ')) continue;
    const c = line.split('|').slice(1, -1).map(x => x.trim());
    homeDesc = c[3] || '';
  }
  check('LD-1-seo-draft-home-desc', homeDesc.length > 20, `len=${homeDesc.length}`);
}

// LD-2: WebSite required schema.org fields + NO bogus SearchAction (public site has no /?q= search URL)
{
  const ws = websiteJsonLd({desc: homeDesc});
  const ok = ws['@context'] === 'https://schema.org' &&
    ws['@type'] === 'WebSite' &&
    ws.name === 'NastyPlot' &&
    ws.url === SITE_ORIGIN + '/' &&
    ws.description === homeDesc &&
    !('potentialAction' in ws);
  check('LD-2-website-schema-no-searchaction', ok, JSON.stringify(ws));
}

// LD-3: BreadcrumbList required schema.org fields for /pokemon/&lt;name&gt;
{
  const bc = pokemonBreadcrumbJsonLd('Charizard-Mega-Y', 'Mega Charizard Y');
  const items = bc && bc.itemListElement;
  const ok = bc &&
    bc['@context'] === 'https://schema.org' &&
    bc['@type'] === 'BreadcrumbList' &&
    Array.isArray(items) && items.length === 3 &&
    items.every((it, i) => it['@type'] === 'ListItem' && it.position === i + 1 && typeof it.name === 'string' && it.name.length > 0 && isHttpsUrl(it.item)) &&
    items[0].item === 'https://nastyplot.org/' &&
    items[1].item === 'https://www.google.com/url?sa=E\&source=gmail\&q=https://nastyplot.org/meta' &&
    items[2].name === 'Mega Charizard Y' &&
    items[2].item === 'https://nastyplot.org/pokemon/Charizard-Mega-Y';
  check('LD-3-pokemon-breadcrumb-schema', ok, JSON.stringify(items));
}

// LD-4: Combined @graph on /pokemon/&lt;name&gt; has valid WebSite + BreadcrumbList
{
  const g = pokemonPageGraphJsonLd('Incineroar', 'Incineroar', {desc: homeDesc});
  const ok = g['@context'] === 'https://schema.org' &&
    Array.isArray(g['@graph']) && g['@graph'].length === 2 &&
    g['@graph'][0]['@type'] === 'WebSite' &&
    g['@graph'][1]['@type'] === 'BreadcrumbList' &&
    !('@context' in g['@graph'][0]) && !('@context' in g['@graph'][1]);
  check('LD-4-pokemon-graph-schema', ok);
}

// LD-5: jsonLdScriptTag produces valid JSON inside &lt;script type=&quot;application/ld+json&quot;&gt; and escapes </script
{
  const tag = jsonLdScriptTag(websiteJsonLd({desc: 'Test &lt;/script&gt; safe'}));
  const m = tag.match(/^&lt;script type=&quot;application\/ld\+json&quot; id=&quot;np-jsonld&quot;&gt;([\s\S]+)<\/script>$/);
  let parsed = null;
  if (m) parsed = JSON.parse(m[1]);
  check('LD-5-script-tag-safe-json', !!m && !tag.includes('&lt;/script&gt; safe') && parsed && parsed['@type'] === 'WebSite');
}

// LD-6: app/build.mjs gates JSON-LD strictly behind PUBLIC and adds nothing to body
{
  const b = fs.readFileSync('app/build.mjs', 'utf8');

    b.includes('jsonLdScriptTag(wsLd)') &&
    b.includes('BreadcrumbList') &&

  check('LD-6-build-public-only-no-searchaction', ok);
