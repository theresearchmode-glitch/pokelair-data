export function hashToPath(hash) {
  const raw = String(hash || '').replace(/^#/, '');
  if (!raw || raw === 'home') return '/';
  const eq = raw.indexOf('='), key = eq >= 0 ? raw.slice(0, eq) : raw, val = eq >= 0 ? raw.slice(eq + 1) : '';
  if (key === 'pokemon' && val) return '/pokemon/' + encodeURIComponent(decodeURIComponent(val));
  if (key === 'about' && val === 'priv') return '/privacy';
  if (key === 'news') return '/gifts';
  const r = BY_CODE.get(key);
  if (r) return r.path + (val ? '#' + raw : '');
  return null;
}

export function pathToRoute(pathname) {
  const p = String(pathname || '/').replace(/\/+$/, '') || '/';
  if (p.startsWith('/pokemon/')) return {code: 'pokemon', param: decodeURIComponent(p.slice(9))};
  if (p.startsWith('/move/')) return {code: 'move', param: decodeURIComponent(p.slice(6))};
  if (p.startsWith('/item/')) return {code: 'item', param: decodeURIComponent(p.slice(6))};
  if (p.startsWith('/ability/')) return {code: 'ability', param: decodeURIComponent(p.slice(9))};
  if (p.startsWith('/nature/')) return {code: 'nature', param: decodeURIComponent(p.slice(8))};
  const r = BY_PATH.get(p);
  if (r) return {code: r.code === 'privacy' ? 'about' : r.code, sub: r.code === 'privacy' ? 'priv' : '', route: r};
  return {code: 'notfound'};
}

const escHtml = s => String(s == null ? '' : s).replace(/[&<>"']/g, m => ({'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'})[m]);
const escXml = s => String(s == null ? '' : s).replace(/[&<>'"]/g, m => ({'&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&apos;', '"': '&quot;'})[m]);

export function applySeoHead(html, {title, desc, canonical}) {
  let out = String(html);
  out = out.replace(/&lt;title&gt;[^<]*<\/title>/, `<title>${escHtml(title)}</title>`);
  out = out.replace(/<meta name="description" content="[^"]*">/, `<meta name="description" content="${escHtml(desc)}">`);
  out = out.replace(/<meta property="og:title" content="[^"]*">/, `<meta property="og:title" content="${escHtml(title)}">`);
  out = out.replace(/<meta property="og:description" content="[^"]*">/, `<meta property="og:description" content="${escHtml(desc)}">`);
  const canonTag = `<link rel="canonical" href="${escHtml(canonical)}">`;
  if (/<link rel="canonical"/.test(out)) out = out.replace(/<link rel="canonical" href="[^"]*">/, canonTag);
  else out = out.replace('&lt;/head&gt;', `${canonTag}\n</head>`);
  return out;
}

export function writePublicStaticPages({baseHtml, outDir, mons = [], origin = SITE_ORIGIN}) {
  const written = [];
  const emit = (relPath, title, desc, summary) => {
    const canonical = origin + (relPath === '/' ? '/' : relPath);
    let page = applySeoHead(baseHtml, {title, desc, canonical});
    if (summary && page.includes('&lt;div id=&quot;app&quot;&gt;&lt;/div&gt;')) {
      page = page.replace('&lt;div id=&quot;app&quot;&gt;&lt;/div&gt;', `<div id="app"><main class="np-main" id="main"><h1>${escHtml(title)}</h1><p>${escHtml(summary)}</p></main></div>`);
    }
    const clean = relPath.replace(/^\/+/, '');
    const dest = clean ? path.join(outDir, clean, 'index.html') : path.join(outDir, 'index.html');
    fs.mkdirSync(path.dirname(dest), {recursive: true});
    fs.writeFileSync(dest, page);
    written.push(relPath);
  };
  for (const r of PUBLIC_ROUTES) emit(r.path, r.title, r.desc, r.desc);
  for (const m of mons) {
    const slug = encodeURIComponent(String(m.n).toLowerCase());
    const t = TEMPLATE_ROUTES.pokemon.title.replace('{Pokémon}', m.d || m.n);
    const d = TEMPLATE_ROUTES.pokemon.desc
      .replace('{Pokémon}', m.d || m.n)
      .replace('{usage}', m.usage != null ? String(m.usage) : '0')
      .replace('{win}', m.win != null ? String(m.win) : '50');
    emit(`/pokemon/${slug}`, t, d, d);
  }
  const urls = written.map(p => `  <url><loc>${escXml(origin + p)}</loc></url>`).join('\n');
  fs.writeFileSync(path.join(outDir, 'sitemap.xml'), `<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n${urls}\n</urlset>\n`);
  fs.writeFileSync(path.join(outDir, 'robots.txt'), `User-agent: *\nAllow: /\nSitemap: ${origin}/sitemap.xml\n`);
  return written;
}

// Cloudflare Worker: anonymous feedback endpoint (/api/feedback) for nastyplot.org.
// Accepts ONLY POST JSON {text <= 2000, page <= 40, kind <= 20} + Turnstile token in header `CF-Turnstile-Response` (or `X-Turnstile-Token`).
// Stores ONLY (at, page, kind, text) in D1 — NEVER IP, user-agent, headers, cookies, or viewer IDs.
const ALLOWED_KINDS = new Set(['bug', 'confusing', 'idea', 'like', 'mistake']);

export async function verifyTurnstile(token, secret, fetchImpl = fetch) {
  if (!token || typeof token !== 'string' || !secret) return false;
  const r = await fetchImpl('https://challenges.cloudflare.com/turnstile/v0/siteverify', {
    method: 'POST',
    headers: {'Content-Type': 'application/x-www-form-urlencoded'},
    body: new URLSearchParams({secret, response: token}).toString(),
  });
  if (!r.ok) return false;
  const j = await r.json();
  return !!(j && j.success);
}

export function validateFeedback(body) {
  if (!body || typeof body !== 'object' || Array.isArray(body)) return {error: 'Invalid JSON body'};
  const keys = Object.keys(body);
  if (keys.some(k => k !== 'text' && k !== 'page' && k !== 'kind')) return {error: 'Unexpected fields'};
  const text = typeof body.text === 'string' ? body.text.trim() : '';
  const page = typeof body.page === 'string' ? body.page.trim() : '';
  const kind = typeof body.kind === 'string' ? body.kind.trim() : '';
  if (text.length < 3 || text.length > 2000) return {error: 'text must be 3..2000 chars'};
  if (!page || page.length > 40) return {error: 'page must be 1..40 chars'};
  if (!kind || kind.length > 20 || !ALLOWED_KINDS.has(kind)) return {error: 'invalid kind'};
  return {ok: {text, page, kind}};
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (url.pathname === '/api/feedback') {
      if (request.method !== 'POST') return new Response('Method Not Allowed', {status: 405});
      let body;
      try { body = await request.json(); } catch (e) { return Response.json({error: 'Invalid JSON'}, {status: 400}); }
      const v = validateFeedback(body);
      if (v.error) return Response.json({error: v.error}, {status: 400});
      const token = request.headers.get('CF-Turnstile-Response') || request.headers.get('X-Turnstile-Token') || '';
      const human = await verifyTurnstile(token, env.TURNSTILE_SECRET);
      if (!human) return Response.json({error: 'Verification failed'}, {status: 403});
      const at = new Date().toISOString().slice(0, 10);
      await env.DB.prepare('INSERT INTO feedback (at, page, kind, text) VALUES (?, ?, ?, ?)')
        .bind(at, v.ok.page, v.ok.kind, v.ok.text)
        .run();
      return Response.json({ok: true}, {status: 201});
    }
    if (url.pathname.startsWith('/data/')) {
      const res = await env.ASSETS.fetch(request);
      return res.status === 404 ? new Response('Not Found', {status: 404}) : res;
    }
    return env.ASSETS.fetch(request);
  },
};
