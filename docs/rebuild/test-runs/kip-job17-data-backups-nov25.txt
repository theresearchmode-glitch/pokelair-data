1. Fallback Order per Data Kind:
   - Online + Official Tournament Pipeline (`update.mjs`, behind `BACKUP_SOURCES=1`, OFF by default):
     1) Limitless API (primary)
     2) `sources/victoryroad.mjs` (official Regionals/Internationals/Specials/Worlds — full OTS + W-L standings)
     3) `sources/showdown-ots.mjs` (Showdown Bo3 OTS replays grouped by UTC day — full OTS + game-level pairings)
   - Official Events Pipeline (`official.mjs`): already uses Victory Road + VR Pastes with Liquipedia API cross-check (no change needed).
   - Regulation Detection (`reg.mjs` & `md-watch.mjs`): already checks Limitless + Showdown replays + official news feeds (no change needed).
   - Ladder / Usage Proxy (`ladder.mjs` & `sources.mjs`): Smogon stats + Showdowntier + Pikalytics (OP.GG stays paused until written permission).

2. Detecting "Primary Is Down or Stale" (`sources/fallback.mjs` `primaryStatus()`):
   - Reuses `md-watch.mjs:69`'s exact threshold (`staleDays = 2`):
     - `down`: Limitless `/tournaments` fetch threw an error (`list === null`).
     - `empty`: Limitless returned `[]` for the active regulation.
     - `stale`: newest saved tournament for the active regulation in `data/index.json` is `>= 2` calendar days old.
     - `ok`: Limitless answered with events and newest saved tournament is `< 2` days old -> backup adapters are NOT called.

3. Merging Without Double Counting (`sources/fallback.mjs` `isDuplicateEvent()`):
   - Namespaced IDs: Limitless uses hex IDs (`6abc...`); Victory Road uses `vr_<slug>`; Showdown OTS uses `sd_<sdFmt>_<YYYY-MM-DD>`. No ID collision is possible.
   - Cross-source event dedup: before accepting any backup tournament, `isDuplicateEvent()` checks (a) exact ID match in `index.tournaments` AND (b) normalized title match (stripping "Pokémon", "VGC", "Championship(s)", punctuation) with start date within `±1` day (catches UTC vs local timezone date shifts, e.g., Limitless indexing a Regional on Nov 15 UTC while Victory Road lists Nov 14).
   - Player privacy (`CLAUDE.md` rule 22): adapters NEVER write files to disk; they return raw handles in memory to `update.mjs`, which immediately hashes every player handle through `pid()` before writing `data/t`, `data/t2`, `data/p`, followed by `scrub-names.mjs --check` in `.github/workflows/update.yml`.

4. Strict Regulation Isolation (`formats.mjs` rules):
   - Every backup tournament is filtered through `sameReg({f: t.format || format, d: t.date}, format)` and `!offFormat(t.name)`.
   - Because `sameReg()` calls `regAt()`, any event dated at or after `REG_END['M-C']` (`2026-12-02T01:59:00Z`) is automatically rejected for Reg M-C even if mistagged `M-C`.
   - `showdown-ots.mjs` additionally validates each replay's `formatid` via `sdReg()` so a `regmdbo3` replay can never enter a `M-C` daily bucket.

// Backup source coordinator for update.mjs (Job 17, off by default unless BACKUP_SOURCES=1).
// Reuses md-watch.mjs's 2-day stale alarm threshold and formats.mjs's strict regulation rules (never mixes regulations).
import {sameReg, offFormat} from '../formats.mjs';

const DAY = 864e5;
const normTitle = s => String(s || '').toLowerCase()
  .replace(/pok[eé]mon/g, '')
  .replace(/\bvgc\b/g, '')
  .replace(/championships?/g, '')
  .replace(/[^a-z0-9]+/g, ' ')
  .trim();

// Detect whether the primary source (Limitless) is down, empty, or stale (>= staleDays old, matching md-watch.mjs's 2-day alarm).
export function primaryStatus({list, index, format, today, staleDays = 2}) {
  const todayIso = today || new Date().toISOString().slice(0, 10);
  const tours = (index && index.tournaments) || {};
  const dates = Object.values(tours)
    .filter(t => t && (!format || sameReg(t, format)))
    .map(t => String(t.d || '').slice(0, 10))
    .filter(d => /^\d{4}-\d{2}-\d{2}$/.test(d))
    .sort();
  const newest = dates.length ? dates[dates.length - 1] : null;
  const ageDays = newest ? Math.floor((Date.parse(todayIso) - Date.parse(newest)) / DAY) : null;
  if (list === null) return {ok: false, reason: 'down', newest, ageDays};
  if (Array.isArray(list) && list.length === 0) return {ok: false, reason: 'empty', newest, ageDays};
  if (ageDays === null || ageDays >= staleDays) return {ok: false, reason: 'stale', newest, ageDays};
  return {ok: true, reason: 'ok', newest, ageDays};
}

// Prevent double-counting when the same event appears in two sources (e.g. Limitless + Victory Road):
// matches by exact id, OR normalized event name + date within 1 day (handles UTC vs local timezone shifts).
export function isDuplicateEvent(cand, existingIndex) {
  const tours = (existingIndex && existingIndex.tournaments) || {};
  if (!cand || !cand.id) return true;
  if (tours[cand.id]) return true;
  const cName = normTitle(cand.name);
  const cTime = Date.parse(String(cand.date || '').slice(0, 10) + 'T12:00:00Z');
  if (!cName || !isFinite(cTime)) return false;
  for (const t of Object.values(tours)) {
    if (!t || !t.name) continue;
    const tTime = Date.parse(String(t.d || '').slice(0, 10) + 'T12:00:00Z');
    if (!isFinite(tTime) || Math.abs(cTime - tTime) > DAY) continue;
    const tName = normTitle(t.name);
    if (!tName) continue;
    if (cName === tName || cName.includes(tName) || tName.includes(cName)) return true;
  }
  return false;
}

// Fetch normalized tournaments, standings, and pairings from ranked backup adapters.
// Returns {tournaments, standingsById, pairingsById, used, errors}.
export async function fetchBackupEvents({format, sdFormat, index, adapters, maxFetch = 20, fetchFn, log = console.log}) {
  const tournaments = [], standingsById = new Map(), pairingsById = new Map(), used = [], errors = [];
  const workingIndex = {tournaments: {...((index && index.tournaments) || {})}};
  let remaining = maxFetch;
  for (const adapter of adapters || []) {
    if (remaining <= 0) break;
    try {
      const rawList = await adapter.listTournaments({format, sdFormat, fetchFn});
      const valid = (Array.isArray(rawList) ? rawList : []).filter(t => {
        if (!t || !t.id || !t.date) return false;
        if (!sameReg({f: t.format || format, d: t.date}, format)) return false;
        if (offFormat(t.name)) return false;
        if (isDuplicateEvent(t, workingIndex)) return false;
        return true;
      }).slice(0, remaining);
      let added = 0;
      for (const t of valid) {
        try {
          const s = await adapter.getStandings(t, {fetchFn});
          if (!Array.isArray(s) || !s.length) continue;
          const p = adapter.getPairings ? await adapter.getPairings(t, {fetchFn}) : [];
          tournaments.push(t);
          standingsById.set(t.id, s);
          pairingsById.set(t.id, Array.isArray(p) ? p : []);
          workingIndex.tournaments[t.id] = {d: t.date, p: t.players, name: t.name, f: t.format || format};
          remaining--;
          added++;
        } catch (err) {
          errors.push(`${adapter.name}:${t.id}: ${err.message}`);
        }
      }
      if (added) used.push(`${adapter.name}(${added})`);
    } catch (err) {
      errors.push(`${adapter.name}: ${err.message}`);
      log(`::warning::backup source ${adapter.name} failed: ${err.message}`);
    }
  }
  return {tournaments, standingsById, pairingsById, used, errors};
}

const MONTHS = ['january', 'february', 'march', 'april', 'may', 'june', 'july', 'august', 'september', 'october', 'november', 'december'];

const AMP = String.fromCharCode(38);
const ENT = {amp: AMP, lt: '<', gt: '>', quot: '"', '#039': "'"};
const dec = s => String(s || '')
  .replace(new RegExp(AMP + '#(\\d+);', 'g'), (_, n) => String.fromCharCode(+n))
  .replace(new RegExp(AMP + '(amp|lt|gt|quot|#039);', 'g'), (_, k) => ENT[k] || '');

export function parseVrDate(s) {
  const m = String(s || '').replace(/\s+/g, ' ').match(/^(\d{1,2})(?: ([A-Za-z]+))?(?: ?[–-] ?\d{1,2})?(?: ([A-Za-z]+))? (\d{4})/);
  if (!m) return null;
  const mo = MONTHS.indexOf((m[2] || m[3] || '').toLowerCase());
  if (mo < 0) return null;
  let y = +m[4];
  if (m[2] && m[3] && mo > MONTHS.indexOf(m[3].toLowerCase())) y--;
  return `${y}-${String(mo + 1).padStart(2, '0')}-${String(+m[1]).padStart(2, '0')}T12:00:00.000Z`;
}

export function parseVrRecord(rec) {
  const m = String(rec || '').match(/(\d+)\s*-\s*(\d+)(?:\s*-\s*(\d+))?/);
  return m ? {wins: +m[1], losses: +m[2], ties: m[3] ? +m[3] : 0} : {wins: 0, losses: 0, ties: 0};
}

export function parseVrEventHtml(html) {
  const info = {};
  for (const m of String(html || '').matchAll(/&lt;tr&gt;<th[^>]*>([^<]+)<\/th><th[^>]*>([\s\S]*?)<\/th><\/tr>/g)) {
    info[strip(m[1])] = strip(m[2]);
  }
  const cut = String(html || '').search(/<h[1-4][^>]*>[^<]*(?:<[^>]+>)*[^<]*(Senior|Junior)/);
  const body = cut > 0 ? html.slice(0, cut) : String(html || '');
  const rows = new Map();
  for (const t of body.matchAll(/<table class="infobox2[^"]*">([\s\S]*?)<\/table>/g)) {
    const head = strip((t[1].match(/&lt;thead&gt;([\s\S]*?)<\/thead>/) || [])[1]);
    if (!/Player/.test(head) || !/Team/.test(head) || !/OTS/.test(head)) continue;
    for (const r of t[1].matchAll(/&lt;tr&gt;([\s\S]*?)<\/tr>/g)) {
      const td = [...r[1].matchAll(/<td[^>]*>([\s\S]*?)<\/td>/g)].map(x => x[1]);
      if (td.length < 7) continue;
      const st = parseInt(strip(td[0]), 10);
      if (!(st > 0) || rows.has(st)) continue;
      rows.set(st, {
        st,
        rec: strip(td[1]),
        player: strip(td[3].split(/<br\s*\/?>/)[0]),
        paste: ((td[6].match(/vrpastes\.com\/([A-Za-z0-9_-]+)/) || [])[1]) || null,
      });
    }
  }
  const ma = s => { const m = String(s || '').match(/([\d,]+)\s*(?:players\s*)?\(?MA\b/); return m ? +m[1].replace(/,/g, '') : null; };
  const regMatch = String(info.Season || '').match(/Regulation(?:\s+Set)?\s+([A-Z]-[A-Z])\b/i);
  return {
    date: parseVrDate(info.Date || ''),
    players: ma(info.Attendance) || ma(info.Capacity) || rows.size,
    game: info.Videogame || '',
    format: regMatch ? regMatch[1].toUpperCase() : '',
    rows: [...rows.values()].sort((a, b) => a.st - b.st),
  };
}

export function parseVrPasteToDecklist(j) {

    name: String(t.species || t.name || ''),
    item: String(t.item || ''),
    ability: String(t.ability || ''),
    attacks: Array.isArray(t.moves) ? t.moves.filter(Boolean).slice(0, 4).map(String) : [],
    tera: String(t.teraType || t.tera || ''),
    nature: String(t.nature || ''),
  })).filter(m => m.name);
}

// Parse Showdown's packed |showteam|p1|... string into update.mjs decklist entries.
// Packed format per mon: Species|Nickname|Item|Ability|Moves|Nature|EVs|Gender|IVs|Shiny|Level|Misc (',,,,,Tera')
export function parseShowteam(packed) {
  if (!packed) return [];
  return String(packed).split(']').map(chunk => {
    const f = chunk.split('|');
    const name = (f[0] || f[1] || '').trim();
    if (!name) return null;
    const item = (f[2] || '').trim();
    const ability = (f[3] || '').trim();
    const attacks = (f[4] || '').split(',').map(x => x.trim()).filter(Boolean).slice(0, 4);
    const nature = (f[5] || '').trim();
    const misc = (f[11] || '').split(',');
    const tera = (misc[5] || '').trim();
    return {name, item, ability, attacks, tera, nature};
  }).filter(Boolean).slice(0, 6);
}

export function parseReplayLog(replay) {
  const log = String((replay && replay.log) || '');
  const players = {p1: '', p2: ''};
  const teams = {p1: [], p2: []};
  let winner = null;
  for (const line of log.split('\n')) {
    if (line.startsWith('|player|p1|')) { const n = line.split('|')[3]; if (n && !players.p1) players.p1 = n.trim(); }
    else if (line.startsWith('|player|p2|')) { const n = line.split('|')[3]; if (n && !players.p2) players.p2 = n.trim(); }
    else if (line.startsWith('|showteam|p1|')) teams.p1 = parseShowteam(line.slice('|showteam|p1|'.length));
    else if (line.startsWith('|showteam|p2|')) teams.p2 = parseShowteam(line.slice('|showteam|p2|'.length));
    else if (line.startsWith('|win|')) winner = line.slice('|win|'.length).trim();
  }
  if (!players.p1 && Array.isArray(replay && replay.players)) players.p1 = replay.players[0] || '';
  if (!players.p2 && Array.isArray(replay && replay.players)) players.p2 = replay.players[1] || '';
  if (!players.p1 || !players.p2 || teams.p1.length !== 6 || teams.p2.length !== 6) return null;
  return {p1: players.p1, p2: players.p2, t1: teams.p1, t2: teams.p2, winner};
}

export const showdownOtsAdapter = {
  name: 'showdown-ots',
  async listTournaments({format, sdFormat, fetchFn}) {
    const baseFmt = sdFormat || '';
    if (!baseFmt) return [];
    const otsFmt = baseFmt.endsWith('bo3') ? baseFmt : `${baseFmt}bo3`;
    const fmtReg = sdReg(baseFmt.replace(/bo3$/, ''));
    if (format && fmtReg && fmtReg !== format) return [];
    const rows = await fetchJson(`${BASE}/search.json?format=${encodeURIComponent(otsFmt)}`, fetchFn);
    if (!Array.isArray(rows)) return [];
    const byDay = new Map();
    for (const r of rows) {
      if (!r || !r.id || !r.uploadtime) continue;
      // Verify each replay's own formatid matches the requested regulation
      const rFmt = String(r.formatid || otsFmt).replace(/bo3$/, '');
      if (format && sdReg(rFmt) && sdReg(rFmt) !== format) continue;
      const day = new Date(+r.uploadtime * 1000).toISOString().slice(0, 10);
      if (!byDay.has(day)) byDay.set(day, []);
      byDay.get(day).push(r);
    }
    return [...byDay.entries()].sort((a, b) => (a[0] < b[0] ? 1 : -1)).map(([day, replays]) => ({
      id: `sd_${otsFmt}_${day}`,
      date: `${day}T23:59:00.000Z`,
      players: new Set(replays.flatMap(r => Array.isArray(r.players) ? r.players : [])).size,
      name: `Showdown OTS Daily ${day}`,
      format: format || fmtReg || '',
      _replays: replays,
    }));
  },
  async getStandings(t, {fetchFn}) {
    const byPlayer = new Map();
    const pairings = [];
    let round = 0;
    for (const r of (t._replays || [])) {
      try {
        const full = await fetchJson(`${BASE}/${encodeURIComponent(r.id)}.json`, fetchFn);
        const parsed = parseReplayLog(full);
        if (!parsed) continue;
        round++;
        const ensure = (pl, dk) => {
          if (!byPlayer.has(pl)) byPlayer.set(pl, {player: pl, placing: null, record: {wins: 0, losses: 0, ties: 0}, decklist: dk});
          return byPlayer.get(pl);
        };
        const a = ensure(parsed.p1, parsed.t1);
        const b = ensure(parsed.p2, parsed.t2);
        if (parsed.winner === parsed.p1) { a.record.wins++; b.record.losses++; }
        else if (parsed.winner === parsed.p2) { b.record.wins++; a.record.losses++; }
        else { a.record.ties++; b.record.ties++; }
        pairings.push({round, player1: parsed.p1, player2: parsed.p2, winner: parsed.winner || '0'});
      } catch (e) {
        // Skip unreadable individual replay
      }
    }
    t._pairings = pairings;
    const sorted = [...byPlayer.values()].sort((x, y) => (y.record.wins - x.record.wins) || (x.record.losses - y.record.losses));
    sorted.forEach((s, i) => { s.placing = i + 1; });
    return sorted;
  },
  async getPairings(t) {
    return Array.isArray(t._pairings) ? t._pairings : [];
  },
};

// Fixture-based tests (no network, fake handles only) for Job 17 backup data sources.
// S7: every check fails on base 39ce8205 (no sources/*.mjs or update.mjs wiring) and passes after patch.
// Usage: node experiments/backup-sources-test.mjs
import fs from 'node:fs';
import assert from 'node:assert/strict';
import {fileURLToPath} from 'node:url';
import {primaryStatus, isDuplicateEvent, fetchBackupEvents} from '../sources/fallback.mjs';
import {victoryRoadAdapter} from '../sources/victoryroad.mjs';
import {showdownOtsAdapter} from '../sources/showdown-ots.mjs';

const ROOT = fileURLToPath(new URL('../', import.meta.url));
const FIX = JSON.parse(fs.readFileSync(ROOT + 'experiments/fixtures/backup-sources.json', 'utf8'));

function mockFetch(url) {
  const u = String(url);
  const ok = body => Promise.resolve({ok: true, status: 200, json: async () => body, text: async () => JSON.stringify(body)});
  if (u.includes('victoryroad.pro/wp-json/wp/v2/pages?search=')) return ok(FIX.vrPagesList);
  const slugM = u.match(/victoryroad\.pro\/wp-json\/wp\/v2\/pages\?slug=([^&]+)/);
  if (slugM) return ok(FIX.vrPagesBySlug[decodeURIComponent(slugM[1])] || []);
  const pasteM = u.match(/vrpaste-backend\.vercel\.app\/api\/paste\/([^?]+)/);
  if (pasteM) return ok(FIX.vrPastes[decodeURIComponent(pasteM[1])] || null);
  if (u.includes('replay.pokemonshowdown.com/search.json')) return ok(FIX.sdSearch);
  const repM = u.match(/replay\.pokemonshowdown\.com\/([^/]+)\.json$/);
  if (repM) return ok(FIX.sdReplays[decodeURIComponent(repM[1])] || null);
  return Promise.resolve({ok: false, status: 404, json: async () => ({})});
}

// 1. Primary down / empty / stale detection (matches md-watch.mjs 2-day alarm)
await test('1. primaryStatus detects down, empty, stale (>=2d), and ok (<=1d)', async () => {
  const idxFresh = {tournaments: {a: {d: '2026-11-23T12:00:00Z', f: 'M-C'}}};
  const idxStale = {tournaments: {a: {d: '2026-11-21T12:00:00Z', f: 'M-C'}}};
  assert.equal(primaryStatus({list: null, index: idxFresh, format: 'M-C', today: '2026-11-24'}).reason, 'down');
  assert.equal(primaryStatus({list: [], index: idxFresh, format: 'M-C', today: '2026-11-24'}).reason, 'empty');
  assert.equal(primaryStatus({list: [{id: 'x'}], index: idxStale, format: 'M-C', today: '2026-11-24'}).reason, 'stale');
  assert.equal(primaryStatus({list: [{id: 'x'}], index: idxFresh, format: 'M-C', today: '2026-11-24'}).ok, true);
});

// 2. Victory Road adapter: exact update.mjs shapes + regulation filter
await test('2. victoryRoadAdapter returns update.mjs shapes and excludes Reg M-D when active is M-C', async () => {
  const list = await victoryRoadAdapter.listTournaments({format: 'M-C', fetchFn: mockFetch, season: 2027});
  assert.equal(list.length, 2, 'should include the 2 M-C regionals and exclude M-D');
  assert.ok(list.every(t => t.format === 'M-C' && t.id.startsWith('vr_')));
  const sample = list.find(t => t.id === 'vr_2027-sample-city-regional');
  assert.ok(sample);
  const standings = await victoryRoadAdapter.getStandings(sample, {fetchFn: mockFetch});
  assert.equal(standings.length, 2);
  assert.equal(standings[0].placing, 1);
  assert.deepEqual(standings[0].record, {wins: 14, losses: 2, ties: 0});
  assert.equal(standings[0].decklist.length, 6);
  assert.equal(standings[0].decklist[0].attacks.length, 4);
  const pairings = await victoryRoadAdapter.getPairings(sample);
  assert.deepEqual(pairings, []);
});
