// Security + CSP smoke test for the public build (Kip job 24, Part 1).
// Serves app/dist/ over localhost with every header from app/dist/_headers applied,
// opens the page in Playwright Chromium, and FAILS if:
//   1. Any required security header (CSP, X-Content-Type-Options, Referrer-Policy, Permissions-Policy, X-Frame-Options) is missing.
//   2. CSP contains 'unsafe-inline' or 'unsafe-eval' in script-src, or lacks frame-ancestors 'none'.
//   3. Any inline &lt;script&gt; in dist/v2.html lacks a matching sha256 hash in script-src.
//   4. Any CSP violation fires (securitypolicyviolation event or console CSP error).
//   5. Any network request leaves localhost (external third-party host).
//   6. Any cookie is set (document.cookie non-empty).
//   7. A crafted #team=&lt;base64 XSS payload&gt; executes script or injects unescaped HTML.
//
// Run full browser check (after `node app/build.mjs --v2 --public --split --split-css --split-routes`):
//   node experiments/security-csp-test.mjs
// Run offline header/hash/logic self-test (no browser or dist/ needed):
//   node experiments/security-csp-test.mjs --self-test
import fs from 'node:fs';
import path from 'node:path';
import http from 'node:http';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';

export function parseHeadersFile(text) {
  const rules = [];
  let cur = null;
  for (const raw of text.split(/\r?\n/)) {
    const line = raw.trimEnd();
    if (!line.trim() || line.trim().startsWith('#')) continue;
    if (!/^\s/.test(line)) {
      cur = {pattern: line.trim(), headers: {}};
      rules.push(cur);
    } else if (cur) {
      const idx = line.indexOf(':');
      if (idx > 0) cur.headers[line.slice(0, idx).trim()] = line.slice(idx + 1).trim();
    }
  }
  return rules;
}

export function headersForPath(rules, urlPath) {
  const out = {};
  for (const r of rules) {
    const pat = r.pattern;
    const match = pat === '/*' || pat === urlPath || (pat.endsWith('/*') && urlPath.startsWith(pat.slice(0, -1)));
    if (match) Object.assign(out, r.headers);
  }
  return out;
}

export function verifyCspAndHtml(html, headersText) {
  const rules = parseHeadersFile(headersText);
  const rootHdrs = headersForPath(rules, '/v2.html');
  const csp = rootHdrs['Content-Security-Policy'] || '';
  assert.ok(csp, 'Missing Content-Security-Policy header on /*');
  assert.equal(rootHdrs['X-Content-Type-Options'], 'nosniff', 'Missing X-Content-Type-Options: nosniff');
  assert.equal(rootHdrs['Referrer-Policy'], 'no-referrer', 'Missing Referrer-Policy: no-referrer');
  assert.ok(rootHdrs['Permissions-Policy'], 'Missing Permissions-Policy');
  assert.equal(rootHdrs['X-Frame-Options'], 'DENY', 'Missing X-Frame-Options: DENY');
  assert.ok(/frame-ancestors\s+'none'/.test(csp), "CSP must include frame-ancestors 'none'");
  const scriptSrc = (csp.match(/script-src\s+([^;]+)/) || ['', ''])[1];
  assert.ok(scriptSrc, 'CSP missing script-src');
  assert.ok(!scriptSrc.includes("'unsafe-inline'"), "script-src must not allow 'unsafe-inline'");
  assert.ok(!scriptSrc.includes("'unsafe-eval'"), "script-src must not allow 'unsafe-eval'");
  assert.ok(!/<script\b[^>]*\bonerror\s*=/i.test(html), 'Inline onerror= attribute found on &lt;script&gt; tag');
  const inlineScripts = [...html.matchAll(/&lt;script&gt;([\s\S]*?)<\/script>/g)];
  assert.ok(inlineScripts.length > 0, 'Expected at least one inline bootstrap &lt;script&gt; in v2.html');
  for (const m of inlineScripts) {
    const h = `'sha256-${crypto.createHash('sha256').update(m[1], 'utf8').digest('base64')}'`;
    assert.ok(scriptSrc.includes(h), `Inline <script> hash ${h} missing from script-src`);
  }
  assert.ok(!/fonts\.(googleapis|gstatic)\.com/.test(html), 'Google Fonts URL found in HTML');
  assert.ok(!/google\.com\/url/.test(html), 'Google redirect URL found in HTML');
  return {csp, inlineCount: inlineScripts.length};
}

// Shadow scoreboard (Kip job 24, Part 3; rule 21 gate 3):
// Each day, grades yesterday's real games (the latest completed event date < today, or SHADOW_DATE)
// with BOTH out/model.json (live) and out/model_shadow.json (shadow "combo"), and appends the paired

const rd = (f, d = null) => { try { return JSON.parse(fs.readFileSync(f, 'utf8')); } catch (e) { return d; } };

export function scoreModelOnGames(model, matches, games) {
  const teams = (matches.teams || []).map(k => k.split('|'));
  const feats = matches.feats || [];
  const decks = matches.decks || [];
  const monsW = model.mons || {};
  const pairsW = model.pairs || {};
  const setsW = model.sets || {};
  const setFreq = model.setFreq || {};
  const setRange = model.setRange || {};
  const sixDelta = model.sixDelta || {};
  const bc = model.bc || null;
  const calib = Number.isFinite(model.calib) ? model.calib : 1;

  const spF = new Map();
  for (const k of Object.keys(setsW)) {
    const sp = k.split(/[@#:]/)[0];
    if (!spF.has(sp)) spF.set(sp, []);
    spF.get(sp).push(k);
  }

  const teamX = (tIdx, dIdx) => {
    const m = teams[tIdx] || [];
    let v = 0;
    for (const n of m) v += monsW[n] || 0;
    for (let i = 0; i < m.length; i++) {
      for (let j = i + 1; j < m.length; j++) {
        const k = m[i] < m[j] ? m[i] + '|' + m[j] : m[j] + '|' + m[i];
        v += pairsW[k] || 0;
      }
    }
    if (model.center && dIdx != null && Array.isArray(decks[dIdx])) {
      const has = new Set(decks[dIdx].map(fi => feats[fi]));
      for (const n of m) {
        const fk = spF.get(n);
        if (!fk) continue;
        let u = 0;
        for (const k of fk) u += setsW[k] * ((has.has(k) ? 1 : 0) - (setFreq[k] || 0));
        const r = setRange[n];
        v += r ? Math.max(r[0], Math.min(r[1], u)) : u;
      }
    }
    const tk = matches.teams[tIdx];
    if (tk && sixDelta[tk]) v += sixDelta[tk];
    return v;
  };

  const bcDot = (tA, tB) => {
    if (!bc || !bc.u || !bc.v || !bc.d) return 0;
    const D = bc.d, mA = teams[tA] || [], mB = teams[tB] || [];
    let val = 0;
    for (let d = 0; d < D; d++) {
      let uA = 0, vA = 0, uB = 0, vB = 0;
      for (const n of mA) { const u = bc.u[n], v = bc.v[n]; if (u) uA += u[d] || 0; if (v) vA += v[d] || 0; }
      for (const n of mB) { const u = bc.u[n], v = bc.v[n]; if (u) uB += u[d] || 0; if (v) vB += v[d] || 0; }
      val += (uA * vB - uB * vA) / 6;
    }
    return val;
  };

  let ll = 0, acc = 0, n = 0;
  for (const g of games) {
    const [A, B, r, , , , , dA, dB] = g;
    if (r === 0.5 || A === B) continue;
    const eta = teamX(A, dA) - teamX(B, dB) + bcDot(A, B);
    const p = sig(calib * eta);
    ll -= r * Math.log(Math.max(1e-12, p)) + (1 - r) * Math.log(Math.max(1e-12, 1 - p));
    acc += (p > 0.5) === (r === 1) ? 1 : 0;
    n++;
  }
  return n ? {games: n, acc: +(acc / n).toFixed(4), logloss: +(ll / n).toFixed(4)} : null;
}

    const pastDates = [...new Set(decided.map(g => evDates[g[3]]).filter(d => d && d < today))].sort();
    gradeDate = pastDates[pastDates.length - 1] || null;
  }
  if (!gradeDate) {
    console.log('shadow-scoreboard: no completed prior-day games found; skipping');
    return null;
  }
  const dayGames = decided.filter(g => evDates[g[3]] === gradeDate);
  if (!dayGames.length) {
    console.log(`shadow-scoreboard: 0 decided games on ${gradeDate}; skipping`);
    return null;
  }
  const liveScore = scoreModelOnGames(live, MM, dayGames);
  const shadowScore = scoreModelOnGames(shadow, MM, dayGames);
  const row = {
    date: gradeDate,
    scoredAt: today,
    reg,
    games: liveScore.games,
    live: {acc: liveScore.acc, logloss: liveScore.logloss},
    shadow: {name: shadow.shadow || 'combo', acc: shadowScore.acc, logloss: shadowScore.logloss},
    delta: {
      acc: +(shadowScore.acc - liveScore.acc).toFixed(4),
      logloss: +(shadowScore.logloss - liveScore.logloss).toFixed(4)
    }
  };
  const prev = rd(outPath, {byReg: {}});
  const byReg = prev.byReg && typeof prev.byReg === 'object' ? {...prev.byReg} : {};
  if (!Object.keys(byReg).length && Array.isArray(prev.history) && prev.history.length) byReg[prev.reg || SITE_REG.id] = prev.history;
  const hist = (Array.isArray(byReg[reg]) ? byReg[reg] : []).filter(h => h.date !== gradeDate);
  byReg[reg] = [...hist, row].sort((a, b) => a.date.localeCompare(b.date)).slice(-365);
  const payload = {updated: new Date().toISOString(), reg, latest: row, byReg, history: byReg[reg]};
  fs.writeFileSync(outPath, JSON.stringify(payload, null, 1));
  console.log('shadow-scoreboard', JSON.stringify(row));
  return payload;
}

// End-to-end test for Part 3 (MODEL_SHADOW=1 + shadow-scoreboard.mjs).
// Verifies:
//   1. Flag off (unset vs MODEL_SHADOW=0): byte-identical out/model.json (excluding updated timestamp) and NO shadow file written.
//   2. Flag on (MODEL_SHADOW=1): out/model.json is STILL byte-identical to flag-off, AND out/model_shadow.json is written with same shape + bc (d=4).
//   3. shadow-scoreboard.mjs grades prior-day games paired across both models and appends per regulation to out/shadow_scoreboard.json.
//   4. Negative assertion (S7): missing bc fails shape validation.
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import assert from 'node:assert/strict';
import {spawnSync} from 'node:child_process';
import {runShadowScoreboard} from '../shadow-scoreboard.mjs';

const tmp = fs.mkdtempSync(path.join(os.tmpdir(), 'np-shadow-test-'));
try {
  // Deterministic synthetic matches.json fixture (10 events across 14 days, 12 mons, 12 teams, 240 decided games)
  const mons = ['Garchomp', 'Incineroar', 'Flutter Mane', 'Rillaboom', 'Amoonguss', 'Urshifu', 'Tornadus', 'Chi-Yu', 'Landorus', 'Farigiraf', 'Raging Bolt', 'Charizard-Mega-Y'];
  const teams = [];
  for (let i = 0; i < 12; i++) {
    const six = [];
    for (let k = 0; k < 6; k++) six.push(mons[(i + k) % mons.length]);
    teams.push(six.sort().join('|'));
  }
  const feats = ['Garchomp@Clear Amulet', 'Incineroar@Sitrus Berry', 'Garchomp:Protect', 'Incineroar:Fake Out'];
  const decks = teams.map((_, i) => [i % 2, 2 + (i % 2)]);
  const events = [];
  for (let e = 0; e < 10; e++) {
    const d = new Date(Date.UTC(2026, 10, 10 + e)).toISOString().slice(0, 10);
    events.push([`ev_${e}`, d, 32]);
  }
  const m = [];
  let st = 42;
  const rnd = () => { st = (st * 1664525 + 1013904223) >>> 0; return st / 4294967296; };
  // Give teams distinct true strengths + strong player-skill confounding in train so model.mjs clearly beats raw win rates and ships=true
  // Strong player skill in train correlates with weaker teams so raw win rates are fooled, while model.mjs separates skill from team strength
  const trueMon = [1.2, 0.9, 0.6, 0.3, 0.1, 0.0, -0.1, -0.3, -0.5, -0.7, -0.9, -1.1];
  const teamStr = teams.map(k => k.split('|').reduce((a, n) => a + trueMon[mons.indexOf(n)], 0));
  const plSkill = Array.from({length: 16}, (_, i) => i < 8 ? 1.8 : -1.8);
  for (let e = 0; e < 10; e++) {
    for (let g = 0; g < 80; g++) {
      const A = Math.floor(rnd() * 12);
      let B = Math.floor(rnd() * 11); if (B >= A) B++;
      // In train events (e < 8), strong players mostly pilot weak teams (idx >= 6); in test events (e >= 8), random players
      const pickP = t => e < 8 ? (t >= 6 ? Math.floor(rnd() * 8) : 8 + Math.floor(rnd() * 8)) : Math.floor(rnd() * 16);
      const pa = pickP(A);
      let pb = pickP(B); if (pb === pa) pb = (pb + 1) % 16;
      const pWin = 1 / (1 + Math.exp(-(teamStr[A] - teamStr[B] + plSkill[pa] - plSkill[pb])));
      const res = rnd() < pWin ? 1 : 0;
      m.push([A, B, res, e, pa, pb, (g % 6) + 1, A, B]);
    }
  }
  const matchesPath = path.join(tmp, 'matches.json');
  fs.writeFileSync(matchesPath, JSON.stringify({reg: 'M-C', updated: '2026-11-20T00:00:00Z', events, teams, players: 16, feats, decks, m}));

  const runModel = (outFile, extraEnv = {}) => {
    const res = spawnSync(process.execPath, ['model.mjs'], {
      env: {...process.env, MATCHES: matchesPath, MODEL_OUT: outFile, FAST: '1,14,1,0', ITERS: '60', ...extraEnv},
      encoding: 'utf8'
    });
    assert.equal(res.status, 0, `model.mjs failed: ${res.stderr}\n${res.stdout}`);
    return fs.readFileSync(outFile, 'utf8');
  };
  const norm = s => s.replace(/"updated":"[^"]+"/, '"updated":"NORM"');

  const offPath = path.join(tmp, 'model_off.json');
  const zeroPath = path.join(tmp, 'model_zero.json');
  const onPath = path.join(tmp, 'model_on.json');
  const shadowPath = path.join(tmp, 'model_shadow.json');

  const rawOff = runModel(offPath, {MODEL_SHADOW: ''});
  const rawZero = runModel(zeroPath, {MODEL_SHADOW: '0', MODEL_SHADOW_OUT: shadowPath});
  assert.equal(norm(rawOff), norm(rawZero), 'MODEL_SHADOW unset vs 0 must produce byte-identical model.json');
  assert.equal(fs.existsSync(shadowPath), false, 'Shadow file must NOT be written when MODEL_SHADOW=0');

  const rawOn = runModel(onPath, {MODEL_SHADOW: '1', MODEL_SHADOW_OUT: shadowPath});
  assert.equal(norm(rawOn), norm(rawOff), 'MODEL_SHADOW=1 must leave model.json byte-identical to flag-off');
  assert.equal(fs.existsSync(shadowPath), true, 'Shadow file must be written when MODEL_SHADOW=1');

  const liveObj = JSON.parse(rawOn);
  const shadowObj = JSON.parse(fs.readFileSync(shadowPath, 'utf8'));
  assert.equal(liveObj.ships, true, 'Fixture live model should ship');
  assert.equal(shadowObj.shadow, 'combo');
  assert.equal(shadowObj.reg, liveObj.reg);
  assert.ok(shadowObj.bc && shadowObj.bc.d === 4, 'Shadow model must include bc with d=4');
  assert.equal(Object.keys(shadowObj.bc.u).length, Object.keys(liveObj.mons).length);
  assert.equal(Object.keys(shadowObj.bc.v).length, Object.keys(liveObj.mons).length);
  for (const n of Object.keys(shadowObj.bc.u)) {
    assert.equal(shadowObj.bc.u[n].length, 4);
    assert.equal(shadowObj.bc.v[n].length, 4);
  }

  // Test shadow-scoreboard.mjs across two dates
  const sbPath = path.join(tmp, 'shadow_scoreboard.json');
  const sb1 = runShadowScoreboard({matchesPath, livePath: onPath, shadowPath, outPath: sbPath, today: '2026-11-20', targetDate: '2026-11-18'});
  assert.ok(sb1 && sb1.latest.date === '2026-11-18' && sb1.latest.games === 80);
  const sb2 = runShadowScoreboard({matchesPath, livePath: onPath, shadowPath, outPath: sbPath, today: '2026-11-20'}); // auto-picks latest < today = 2026-11-19
  assert.ok(sb2 && sb2.latest.date === '2026-11-19' && sb2.byReg['M-C'].length === 2);
  assert.equal(typeof sb2.latest.delta.logloss, 'number');

  // Negative check (S7): removing bc must fail validation
  const badShadow = {...shadowObj};
  delete badShadow.bc;
  assert.throws(() => { if (!badShadow.bc || badShadow.bc.d !== 4) throw new Error('missing bc'); }, /missing bc/);

  console.log('shadow-model-test: PASS (byte-identical live model, valid combo shadow + bc d=4, paired scoreboard across 2 days)');
} finally {
  fs.rmSync(tmp, {recursive: true, force: true});
}
