export function judgeShadowScoreboard(history, {
  minDays = 14,
  minGames = 8000,
  alpha = 0.05,
  bootIters = 2000,
  seed = 2026
} = {}) {
  const rows = (history || []).filter(r => r && r.games > 0 && Number.isFinite(r.delta?.logloss));
  const totalGames = rows.reduce((a, r) => a + r.games, 0);
  if (rows.length < minDays || totalGames < minGames) {
    return { status: 'UNKNOWN', reason: `need >=${minDays} days and >=${minGames} games (have ${rows.length}d, ${totalGames}g)`, days: rows.length, totalGames };
  }
  const wMean = arr => {
    let sw = 0, sx = 0;
    for (const r of arr) { sw += r.games; sx += r.games * r.delta.logloss; }
    return sx / sw;
  };
  const mean = wMean(rows);

  let st = seed >>> 0;
  const rnd = () => { st = (st * 1664525 + 1013904223) >>> 0; return st / 4294967296; };
  const boots = new Float64Array(bootIters);
  for (let b = 0; b < bootIters; b++) {
    let sw = 0, sx = 0;
    for (let i = 0; i < rows.length; i++) {
      const r = rows[Math.floor(rnd() * rows.length)];
      sw += r.games; sx += r.games * r.delta.logloss;
    }
    boots[b] = sx / sw;
  }
  boots.sort();
  const ciLo = boots[Math.floor((alpha / 2) * bootIters)];
  const ciHi = boots[Math.floor((1 - alpha / 2) * bootIters)];

  let worstLooMean = -Infinity;
  for (let i = 0; i < rows.length; i++) {
    const m = wMean(rows.filter((_, j) => j !== i));
    if (m > worstLooMean) worstLooMean = m;
  }
  const status = ciHi < 0 && worstLooMean < 0 ? 'BETTER' : ciLo > 0 ? 'WORSE' : 'UNKNOWN';
  return { status, days: rows.length, totalGames, meanDeltaLL: +mean.toFixed(5), ci95: [+ciLo.toFixed(5), +ciHi.toFixed(5)], worstLooMean: +worstLooMean.toFixed(5) };
}
